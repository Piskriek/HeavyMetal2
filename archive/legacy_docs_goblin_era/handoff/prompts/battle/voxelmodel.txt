export interface PaletteEntry { name: string; color: readonly [number, number, number] /* 0..1 sRGB */; roughness: number; metalness: number; emissive: number /* 0..1 glow strength */; alpha: number /* 0..1, below 1 = see-through like glass */; material?: string /* optional id of a procedural material preset */ }
export interface VoxelModel { id: string; name: string; size: readonly [number, number, number] /* each 1..96 */; pivot: readonly [number, number, number] /* in voxel units */; palette: PaletteEntry[] /* index i+1 in the grid; 0 = empty; at most 255 */; cells: Uint8Array /* x-fastest then y then z, length x*y*z, 0 = empty */ }

grid.ts (all return NEW models, never mutate; out-of-range coordinates are ignored): createModel(id, name, size), getVoxel(m, x, y, z), setVoxel(m, x, y, z, v), fillBox(m, a, b, v), fillSphere(m, centre, radius, v), fillLine(m, a, b, v, thickness), floodFill(m, x, y, z, v) (6-connected same value), replaceColor(m, from, to), hollow(m) (keep only voxels with an empty or edge neighbour), extrude(m, axis, fromCoord, length), mirror(m, axis: 'x'|'y'|'z') (copies the lower half onto the upper half), flip(m, axis), rotate90(m, axis), translate(m, dx, dy, dz) (wrap off, voxels falling outside are dropped), trim(m) (shrinks size to the occupied bounds and moves the pivot with it), merge(a, b, offset) (b drawn over a, palettes merged and deduplicated, indices remapped), countVoxels(m).

codec.ts: encodeModel(m): string and decodeModel(text): { model: VoxelModel | null; errors: string[] }: JSON with the palette and the cells run-length encoded (pairs of value,count in x-fastest order, then base64 of the bytes) so a detailed model stays small; round trip exact; decode validates and never throws.

validate.ts: validateModel(x: unknown): { ok: boolean; errors: string[] } readable sentences (size limits, palette limit 255, cell values within the palette, colour ranges, cells length), never throws on any input.

mesh.ts: meshModel(m, opts?: { ao?: boolean /* default true */; greedy?: boolean /* default true */ }): { positions: Float32Array; normals: Float32Array; colors: Float32Array /* rgb per vertex, ambient occlusion baked in when ao */; indices: Uint32Array; paletteIndex: Uint8Array /* per vertex, 1-based */; triangleCount: number }. Rules: only faces between a solid voxel and empty space (or the model border) are emitted; transparent voxels (alpha < 1) emit faces towards other transparent or empty voxels but not between two voxels of the same transparent entry; positions in voxel units minus the pivot; counter-clockwise winding seen from outside; classic voxel ambient occlusion per face vertex (0..3 from the two side neighbours and the corner neighbour) darkening colour by 0.25 per level at most; greedy merging of coplanar neighbouring faces with the same palette index, normal and the same four AO values.

stats.ts: modelStats(m): { voxels, bounds: { min, max }, paletteUse: number[] (voxels per palette index), triangles: number } and modelHash(m): string (stable short hash of size, palette and cells).

presets.ts: a few small example models built with the grid functions in code (each returns a VoxelModel): `exampleCube()` (3x3x3 two colours), `exampleBall()` (radius 6 sphere, hollow), `exampleBarrel()` (16 high, with two metal bands using palette entries with metalness 0.9), `exampleCone()` (traffic cone with a white stripe, 12 high).
