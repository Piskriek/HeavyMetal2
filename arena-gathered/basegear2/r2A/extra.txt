// tests/basegear.test.ts
import test from "node:test";
import assert from "node:assert/strict";
import * as THREE from "three";
import {
  createMaterials,
  hardpoint,
  heavyMill,
  bin,
  repeater,
  draftingTable,
  setLamp,
  triangles,
  type Gear,
} from "../src/index";

const m = createMaterials();
const bb = (g: Gear) => new THREE.Box3().setFromObject(g.group);
const has = (g: Gear, ...names: string[]) =>
  names.forEach((n) => assert.ok(g.sockets.some((s) => s.name === n), n));

test("gear fits its cells, stands on the slab, has its sockets and lamps", () => {
  for (const stage of [1, 6]) {
    const o = { stage };
    const all = {
      hardpoint: hardpoint(m, o),
      heavyMill: heavyMill(m, o),
      bin: bin(m, o),
      repeater: repeater(m, o),
      draftingTable: draftingTable(m, o),
    };
    for (const [k, g] of Object.entries(all)) {
      const t = triangles(g);
      const cap = stage === 1 ? 900 : k === "heavyMill" ? 10000 : 8000;
      assert.ok(t > 0 && t <= cap, `${k} s${stage}: ${t}`);
      assert.ok(g.group.children.length <= 8, k);
      assert.ok(bb(g).min.y >= -0.02, `${k} sinks`);
      assert.ok(g.colliders.length > 0, k);
    }
    const h = bb(all.hardpoint);
    assert.ok(
      h.min.x >= -0.01 &&
        h.max.x <= 8.01 &&
        h.min.z >= -0.01 &&
        h.max.z <= 8.01 &&
        h.max.y <= 1.2
    );
    const mill = bb(all.heavyMill);
    assert.ok(mill.max.y >= 3.6 && mill.max.y <= 5.5 && mill.max.x - mill.min.x <= 6);
    for (const k of ["bin", "repeater", "draftingTable"] as const) {
      const b = bb(all[k]);
      assert.ok(b.min.x >= -0.01 && b.max.x <= 4.01 && b.min.z >= -0.01 && b.max.z <= 4.01, k);
    }
    assert.ok(bb(all.repeater).max.y >= 6 && bb(all.draftingTable).max.y <= 1.6);
    has(all.hardpoint, "mount", "power");
    has(all.heavyMill, "vent", "hopper", "power");
    has(all.bin, "link");
    has(all.repeater, "top", "box");
    has(all.draftingTable, "screen");
    for (const g of [all.heavyMill, all.bin, all.repeater, all.draftingTable]) {
      assert.ok(g.lamps.length > 0);
      setLamp(g.lamps[0]!, 1);
    }
  }
});

/* --------------------------- additional checks --------------------------- */

test("materials expose the required set and glass is physical", () => {
  const mats = createMaterials();
  for (const k of [
    "gunmetal",
    "darkSteel",
    "paint",
    "copper",
    "rubber",
    "hazard",
    "concrete",
  ] as const) {
    assert.ok(mats[k] instanceof THREE.MeshStandardMaterial, k);
  }
  assert.ok(mats.glass instanceof THREE.MeshPhysicalMaterial);
  assert.equal(mats.glass.transmission, 0); // no transmission
});

test("every mesh is non-indexed with position/normal/uv and finite coords", () => {
  for (const stage of [1, 6]) {
    const o = { stage };
    const gears = [
      hardpoint(m, o),
      heavyMill(m, o),
      bin(m, o),
      repeater(m, o),
      draftingTable(m, o),
    ];
    for (const g of gears) {
      g.group.traverse((obj) => {
        if (obj instanceof THREE.Mesh) {
          const geo = obj.geometry;
          assert.equal(geo.index, null, "must be non-indexed");
          const pos = geo.getAttribute("position");
          const nrm = geo.getAttribute("normal");
          const uv = geo.getAttribute("uv");
          assert.ok(pos && nrm && uv, "needs position/normal/uv");
          for (let i = 0; i < pos.count; i++) {
            assert.ok(Number.isFinite(pos.getX(i)));
            assert.ok(Number.isFinite(pos.getY(i)));
            assert.ok(Number.isFinite(pos.getZ(i)));
          }
        }
      });
    }
  }
});

test("lamp glow is clamped to emissive 0..1", () => {
  const g = heavyMill(m, { stage: 6 });
  const lamp = g.lamps[0]!;
  setLamp(lamp, 1);
  assert.equal((lamp.material as THREE.MeshStandardMaterial).emissiveIntensity, 1);
  setLamp(lamp, 5);
  assert.equal((lamp.material as THREE.MeshStandardMaterial).emissiveIntensity, 1);
  setLamp(lamp, -2);
  assert.equal((lamp.material as THREE.MeshStandardMaterial).emissiveIntensity, 0);
});

test("colliders are axis-aligned boxes inside the part bounds", () => {
  for (const stage of [1, 6]) {
    const g = bin(m, { stage });
    for (const c of g.colliders) {
      assert.ok(c.min[0] <= c.max[0] && c.min[1] <= c.max[1] && c.min[2] <= c.max[2]);
    }
  }
});
