export interface Terrain { readonly spec: TerrainSpec; readonly heights: Float32Array /* rows*cols, index r*cols+c, node (c,r) is at x=originX+c*cell, z=originZ+r*cell */; readonly surfaceA: Uint8Array; readonly surfaceB: Uint8Array; readonly blend: Uint8Array /* weight of surface B, 0..255 */ }

export interface Brush { readonly kind: BrushKind; readonly x: number; readonly z: number; readonly radius: number; readonly strength: number; readonly falloff: 'smooth' | 'linear' | 'flat'; readonly surface?: number /* paint */; readonly target?: number /* flatten height */ }
export function createTerrain(spec: TerrainSpec, fill?: { height?: number; surface?: number }): Terrain   // cols,rows >= 2 else throw; defaults height 0, surface 0; A == B == surface, blend 0
export function cloneTerrain(t: Terrain): Terrain
export function heightAt(t: Terrain, x: number, z: number): number   // MUST follow the physics collider's triangulation: each cell has nodes a=(c,r) b=(c+1,r) c=(c,r+1) d=(c+1,r+1), the diagonal runs a-d; with u,v in [0,1] the position inside the cell: if u >= v then h = ha + u*(hb-ha) + v*(hd-hb) else h = ha + v*(hc-ha) + u*(hd-hc). Outside the grid: clamp x,z to the grid edge.
export function normalAt(t: Terrain, x: number, z: number): [number, number, number]   // unit normal of the triangle under (x,z) (same triangulation)
export function normalYAtCell(t: Terrain, c: number, r: number): number   // normal.y at node (c,r) from central differences of the neighbouring nodes (clamped at the border): n = normalize(-dhdx, 1, -dhdz) with dhdx = (h(c+1)-h(c-1))/(2*cell) etc., return n.y
export function applyBrush(t: Terrain, b: Brush): DirtyRect | null   // mutates t; returns the node rect it touched (clamped to the grid) or null if the brush is entirely outside the grid or touches no node

export function generateIsland(spec: TerrainSpec, seed: number, o: IslandOptions): Terrain

BRUSH RULES (grid nodes whose distance d to (x,z) is < radius are affected; weight w(d): 'smooth' = 1 - (3t^2 - 2t^3) with t = d/radius; 'linear' = 1 - t; 'flat' = 1; w = 0 for d >= radius):
- raise: h += strength * w. lower: h -= strength * w. flatten: h += (target - h) * clamp(strength*w, 0, 1) (target default 0). smooth: h += (mean of the 8 neighbours (clamped at the border, use the node itself where a neighbour is missing) - h) * clamp(strength*w, 0, 1); read neighbours from a copy so the result does not depend on iteration order.
- paint (strength 0..1 is the dab amount, k = clamp(strength*w, 0, 1), skip nodes where k is 0): each node has surface A, surface B and wB = blend/255. Let S = brush.surface. If S == A: wB = wB*(1-k). Else if S == B: wB = wB + (1-wB)*k. Else (a new surface): if wB < 0.5 then B = S and wB = k; otherwise A = B, B = S, wB = k (the roles swap). Afterwards blend = round(wB*255); if blend == 0 then B = A; if blend == 255 then A = B and blend = 0.
- The dirty rect is the min/max node range inside the radius (clamped), nothing else.

ENCODING: EncodedTerrain = { v: 1; spec: TerrainSpec; h: string; a: string; b: string; w: string }, all strings base64 (use a small own base64 on Uint8Array, no Node Buffer: it must run in the browser). Heights are quantised to Int16 in units of 0.01 m (clamp to +-327.67 m), stored PLANAR (all low bytes of every node first, then all high bytes, so flat ground makes long runs), then run-length encoded as bytes. a, b, w are Uint8Arrays, each run-length encoded as (count 1..255, value) byte pairs. Decoding validates sizes and throws a clear Error on anything malformed (wrong v, wrong lengths, missing fields). A flat 129x129 terrain must encode to < 12 KB of JSON.

GENERATOR: value noise from an integer hash of (ix, iz, seed) (Math.imul based, result in [0,1)), smoothstep interpolation, 5-octave fbm (gain 0.5, lacunarity 2), no trig. Island: d = distance from the grid centre / (radius * half the smaller extent in metres); base height = height * (1 - smoothstep(0.0, 1.0, d))^1.3 shaped so the centre is high, plus (fbm - 0.5) * roughness scaled by that same island mask, minus seaDepth * max(0, d - 1) so the sea floor drops away. Sea level is height 0. Surfaces per node: seabed where h < -0.5; between -0.5 and 1.0 sand (with blend towards seabed below 0 and towards grass above 0.5 using surfaceB and the blend byte); above 1 grass; where normalYAtCell < 0.6 rock (A = rock, B = whatever it was, blend by how steep); where normalYAtCell < 0.35 cliff (A = cliff). Cliff/rock only apply above -0.5 (underwater stays seabed). Process surfaces in a second pass after all heights are final.
